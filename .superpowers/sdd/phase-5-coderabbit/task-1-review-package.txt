+# Phase 5: CodeRabbit Integration Implementation Plan
+
+> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
+
+**Goal:** Integrate CodeRabbit code review platform with Cartograph dependency graph, showing review status on nodes and enabling on-demand refresh of review data.
+
+**Architecture:** Read-only CodeRabbit integration with hybrid caching: fetch all reviews at analysis time (Phase 3) and store in database for instant display, then allow users to refresh individual file reviews on-demand via API endpoint. Reviews cached with org-level RLS; if CodeRabbit unavailable, analysis completes successfully.
+
+**Tech Stack:** Next.js 16 App Router, TypeScript strict, React 19, Supabase (Postgres with RLS), CodeRabbit API (read-only), Jest for testing, Tailwind v4.
+
+**Spec:** `docs/specs/phase-5.md`
+
+## Global Constraints
+
+- Review statuses: `'approved'` | `'needs-review'` | `'issues'` | `'not-reviewed'` (enum in database)
+- Badge colors: `#22c55e` (green-approved), `#eab308` (yellow-needs-review), `#ef4444` (red-issues), `#9ca3af` (gray-not-reviewed)
+- CodeRabbit API timeout: 30 seconds maximum
+- RLS enforced at database level: users see only reviews for repos in their organization
+- Graceful degradation: if CodeRabbit API fails, analysis completes with warnings, reviews show as "not-reviewed"
+- All TypeScript strict mode, zero any, all tests passing before merge
+- File path normalization: reviews keyed by normalized file paths matching parser output
+
+## Review Focus
+
+1. **File path mismatch:** CodeRabbit returns different file path format than parser — test reviews display on correct nodes despite path normalization differences
+2. **RLS bypass:** User from different organization attempts to access/refresh reviews — verify 403 or no-data response, never cross-org leakage
+3. **CodeRabbit timeout:** API call exceeds 30s during analysis fetch — verify analysis completes with warning log, reviews show "not-reviewed", no hanging
+4. **Concurrent refresh:** User clicks "Refresh Reviews" twice rapidly on same file — verify request deduplication or graceful handling, no duplicate API calls
+5. **Empty organization:** Organization has no CodeRabbit API key configured — verify graceful degradation, no 500 errors, analysis proceeds with "not-reviewed" for all files

+```typescript
+// lib/coderabbit/client.ts
+import { ReviewData, ReviewStatus, CodeRabbitReview } from './types';
+
+const CODERABBIT_API_BASE = 'https://api.coderabbit.ai/v1';
+const API_TIMEOUT_MS = 30000; // 30 seconds
+
+function normalizeFilePath(path: string): string {
+  // Normalize paths from CodeRabbit to match parser output
+  return path.replace(/\\/g, '/').replace(/^\/+/, '');
+}
+
+function isValidReviewStatus(status: unknown): status is ReviewStatus {
+  return ['approved', 'needs-review', 'issues', 'not-reviewed'].includes(String(status));
+}
+
+export async function fetchReviewsForRepo(
+  repoUrl: string,
+  apiKey: string
+): Promise<ReviewData[]> {
+  if (!apiKey) {
+    console.warn('CodeRabbit API key not configured');
+    return [];
+  }
+
+  try {
+    const controller = new AbortController();
+    const timeoutId = setTimeout(() => controller.abort(), API_TIMEOUT_MS);
+
+    const response = await fetch(`${CODERABBIT_API_BASE}/reviews`, {
+      method: 'POST',
+      headers: {
+        'Authorization': `Bearer ${apiKey}`,
+        'Content-Type': 'application/json',
+      },
+      body: JSON.stringify({
+        repository_url: repoUrl,
+      }),
+      signal: controller.signal,
+    });
+
+    clearTimeout(timeoutId);
+
+    if (!response.ok) {
+      console.warn(`CodeRabbit API error: ${response.status} ${response.statusText}`);
+      return [];
+    }
+
+    const data = await response.json();
+    const reviews: CodeRabbitReview[] = data.reviews || [];
+
+    return reviews
+      .filter((r) => isValidReviewStatus(r.review_status))
+      .map((review) => ({
+        filePath: normalizeFilePath(review.file_path),
+        reviewStatus: review.review_status as ReviewStatus,
+        commentCount: review.comment_count || 0,
+        summary: review.summary || null,
+        externalReviewUrl: review.external_url || null,
+        fetchedAt: new Date(),
+      }));
+  } catch (error) {
+    if (error instanceof Error && error.name === 'AbortError') {
+      console.warn('CodeRabbit API request timed out (30s)');
+    } else {
+      console.warn('CodeRabbit API request failed:', error);
+    }
+    return [];
+  }
+}
+```

+```typescript
+// lib/coderabbit/storage.ts
+import { createClient } from '@supabase/supabase-js';
+import { ReviewData, ReviewStatus } from './types';
+
+const supabase = createClient(
+  process.env.NEXT_PUBLIC_SUPABASE_URL!,
+  process.env.SUPABASE_SERVICE_KEY!
+);
+
+export async function storeReviews(
+  orgId: string,
+  repoId: string,
+  reviews: ReviewData[]
+): Promise<void> {
+  if (reviews.length === 0) return;
+
+  const rows = reviews.map((review) => ({
+    id: crypto.randomUUID(),
+    organization_id: orgId,
+    repository_id: repoId,
+    file_path: review.filePath,
+    review_status: review.reviewStatus as ReviewStatus,
+    comment_count: review.commentCount,
+    summary: review.summary,
+    external_review_url: review.externalReviewUrl,
+    fetched_at: review.fetchedAt.toISOString(),
+    created_at: new Date().toISOString(),
+  }));
+
+  const { error } = await supabase
+    .from('coderabbit_reviews')
+    .upsert(rows, { onConflict: 'repository_id,file_path' });
+
+  if (error) {
+    console.error('Failed to store reviews:', error);
+    throw error;
+  }
+}
+
+export async function getReviewForFile(

+  if (error) {

+```typescript
+// app/api/coderabbit/refresh/route.ts
+import { NextRequest, NextResponse } from 'next/server';
+import { auth } from '@clerk/nextjs/server';
+import { createClient } from '@supabase/supabase-js';
+import { getReviewForFile } from '@/lib/coderabbit/storage';
+import { fetchReviewsForRepo } from '@/lib/coderabbit/client';
+import type { ReviewData } from '@/lib/coderabbit/types';
+
+export async function POST(request: NextRequest) {
+  try {
+    // 1. Authentication
+    const { userId, sessionClaims } = await auth();
+    if (!userId) {
+      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
+    }
+
+    // 2. Parse request body
+    const { analysisId, filePath } = await request.json();
+    if (!analysisId || !filePath) {
+      return NextResponse.json(
+        { error: 'Missing analysisId or filePath' },
+        { status: 400 }
+      );
+    }
+
+    // 3. Verify analysis ownership via RLS
+    const supabase = createClient(
+      process.env.NEXT_PUBLIC_SUPABASE_URL!,
+      process.env.SUPABASE_SERVICE_KEY!
+    );
+
+    const { data: analysis, error: analysisError } = await supabase
+      .from('analyses')
+      .select('id, repository_id, organization_id')
+      .eq('id', analysisId)
+      .single();
+
+    if (analysisError || !analysis) {
+      return NextResponse.json({ error: 'Analysis not found' }, { status: 404 });
+    }
+
+    // 4. Verify user is in organization (RLS check)
+    const { data: orgMember, error: orgError } = await supabase
+      .from('organization_members')
+      .select('organization_id')
+      .eq('organization_id', analysis.organization_id)
+      .eq('user_id', userId)
+      .single();
+
+    if (orgError || !orgMember) {
+      return NextResponse.json(
+        { error: 'Unauthorized - not in organization' },
+        { status: 403 }
+      );
+    }
+
+    // 5. Get repository for CodeRabbit API call
+    const { data: repo, error: repoError } = await supabase
+      .from('repositories')
+      .select('url, coderabbit_api_key')
+      .eq('id', analysis.repository_id)
+      .single();
+
+    if (repoError || !repo) {
+      return NextResponse.json({ error: 'Repository not found' }, { status: 404 });
+    }
+
+    // 6. Get CodeRabbit API key from organization
+    const { data: org, error: orgKeyError } = await supabase
+      .from('organizations')
+      .select('coderabbit_api_key')
+      .eq('id', analysis.organization_id)
+      .single();
+
+    if (orgKeyError || !org?.coderabbit_api_key) {
+      // Return 503 - CodeRabbit not configured
+      return NextResponse.json(
+        { error: 'CodeRabbit not configured for organization' },
+        { status: 503 }
+      );
+    }
+
+    // 7. Fetch fresh review from CodeRabbit
+    const reviews = await fetchReviewsForRepo(repo.url, org.coderabbit_api_key);
+    const refreshedReview = reviews.find((r) => r.filePath === filePath);
+
+    if (!refreshedReview) {
+      // File has no review (not-reviewed status)
+      const defaultReview: ReviewData = {
+        filePath,
+        reviewStatus: 'not-reviewed',
+        commentCount: 0,
+        summary: null,
+        externalReviewUrl: null,
+        fetchedAt: new Date(),
+      };
+
+      // Store the default review
+      await supabase
+        .from('coderabbit_reviews')
+        .upsert({
+          id: crypto.randomUUID(),
+          organization_id: analysis.organization_id,
+          repository_id: analysis.repository_id,
+          file_path: filePath,
+          review_status: 'not-reviewed',
+          comment_count: 0,
+          summary: null,
+          external_review_url: null,
+          fetched_at: new Date().toISOString(),
+          created_at: new Date().toISOString(),
+        }, { onConflict: 'repository_id,file_path' });
+
+      return NextResponse.json(defaultReview);
+    }
+
+    // 8. Store refreshed review in database
+    await supabase
+      .from('coderabbit_reviews')
+      .upsert({
+        id: crypto.randomUUID(),
+        organization_id: analysis.organization_id,
+        repository_id: analysis.repository_id,
+        file_path: refreshedReview.filePath,
+        review_status: refreshedReview.reviewStatus,
+        comment_count: refreshedReview.commentCount,
+        summary: refreshedReview.summary,
+        external_review_url: refreshedReview.externalReviewUrl,
+        fetched_at: new Date().toISOString(),
+        created_at: new Date().toISOString(),
+      }, { onConflict: 'repository_id,file_path' });
+
+    return NextResponse.json(refreshedReview);
+  } catch (error) {
+    console.error('Refresh endpoint error:', error);
+    return NextResponse.json(
+      { error: 'Internal server error' },
+      { status: 500 }
+    );
+  }
+}
+```

+// hooks/useFileDetail.ts (add review-related state and functions)
+import { useState } from 'react';
+import type { ReviewData } from '@/lib/coderabbit/types';
+
+interface UseFileDetailProps {
+  files: File[];
+  edges: Edge[];
+  analysisId: string;
+  reviews: Map<string, ReviewData>;
+}
+
+export function useFileDetail({
+  files,
+  edges,
+  analysisId,
+  reviews,
+}: UseFileDetailProps) {
+  const [selectedFile, setSelectedFile] = useState<string | null>(null);
+  const [explanation, setExplanation] = useState<string | null>(null);
+  const [isLoadingExplanation, setIsLoadingExplanation] = useState(false);
+  const [reviewData, setReviewData] = useState<ReviewData | null>(null);
+  const [isRefreshingReview, setIsRefreshingReview] = useState(false);
+
+  const handleSelectFile = (filePath: string) => {
+    setSelectedFile(filePath);
+    setExplanation(null); // Clear previous explanation
+
+    // Update review data for selected file
+    const review = reviews.get(filePath);
+    setReviewData(review || null);
+  };
+
+  const handleLoadExplanation = async (filePath: string) => {
+    setIsLoadingExplanation(true);
+    try {
+      const response = await fetch(`/api/explain`, {
+        method: 'POST',
+        headers: { 'Content-Type': 'application/json' },
+        body: JSON.stringify({ analysisId, filePath }),
+      });
+      const data = await response.json();
+      setExplanation(data.explanation);
+    } catch (error) {
+      console.error('Failed to load explanation:', error);
+      setExplanation('Failed to load explanation');
+    } finally {
+      setIsLoadingExplanation(false);
+    }
+  };
+
+  const handleRefreshReview = async (filePath: string) => {
+    setIsRefreshingReview(true);
+    try {
+      const response = await fetch(`/api/coderabbit/refresh`, {
+        method: 'POST',
+        headers: { 'Content-Type': 'application/json' },
+        body: JSON.stringify({ analysisId, filePath }),
+      });
+
+      if (!response.ok) throw new Error('Failed to refresh review');
+
+      const freshReview: ReviewData = await response.json();
+      setReviewData(freshReview);
+
+      // Update the reviews map (caller should update their state)
+      reviews.set(filePath, freshReview);
+    } catch (error) {
+      console.error('Failed to refresh review:', error);
+    } finally {
+      setIsRefreshingReview(false);
+    }
+  };
+
+  // ... existing metric calculations (fanIn, fanOut, blastRadius) ...
+
+  return {
+    selectedFile,
+    onSelectFile: handleSelectFile,
+    explanation,
+    isLoadingExplanation,
+    onLoadExplanation: handleLoadExplanation,
+    reviewData,
+    isRefreshingReview,
+    onRefreshReview: handleRefreshReview,

+// hooks/useAnalysis.ts (update Analysis type and conversion)
+import type { ReviewData } from '@/lib/coderabbit/types';
+
+interface Analysis {
+  id: string;
+  organizationId: string;
+  repositoryUrl: string;
+  fileList: File[];
+  edgeList: Edge[];
+  coverage: { parsed: number; skipped: number; reasons: Record<string, number> };
+  reviews?: ReviewData[]; // From API as array
+}
+
+export function useAnalysis(analysisId: string) {
+  const [data, setData] = useState<Analysis | null>(null);
+  const [loading, setLoading] = useState(true);
+  const [error, setError] = useState<string | null>(null);
+
+  useEffect(() => {
+    const fetchAnalysis = async () => {
+      try {
+        const response = await fetch(`/api/analyses/${analysisId}`);
+        if (!response.ok) throw new Error('Failed to fetch analysis');
+
+        const analysis: Analysis = await response.json();
+        setData(analysis);
+      } catch (err) {
+        setError(err instanceof Error ? err.message : 'Unknown error');
+      } finally {
+        setLoading(false);
+      }
+    };
+
+    fetchAnalysis();
+  }, [analysisId]);
+
+  // Convert reviews array to Map for O(1) lookup by filePath
+  const reviewsMap = useMemo(() => {
+    const map = new Map<string, ReviewData>();
+    (data?.reviews || []).forEach((review) => {
+      map.set(review.filePath, review);
+    });
+    return map;
+  }, [data?.reviews]);
+
+  return {
+    analysis: data,
+    reviews: reviewsMap,
+    loading,
+    error,
+  };
+}
+```

+import type { ReviewData, ReviewStatus, CodeRabbitReview } from './types';
+
+const API_ENDPOINT = 'https://api.coderabbit.ai/v1/reviews';
+const TIMEOUT_MS = 30000;
+
+/**
+ * Valid review statuses.
+ */
+const VALID_REVIEW_STATUSES = new Set<ReviewStatus>([
+  'approved',
+  'needs-review',
+  'issues',
+  'not-reviewed',
+]);
+
+/**
+ * Normalize a file path: convert backslashes to forward slashes,
+ * and strip leading slashes.
+ */
+function normalizeFilePath(path: string): string {
+  return path.replace(/\\/g, '/').replace(/^\/+/, '');
+}
+
+/**
+ * Check if a value is a valid review status.
+ */
+function isValidReviewStatus(status: unknown): status is ReviewStatus {
+  return VALID_REVIEW_STATUSES.has(status as ReviewStatus);
+}
+
+/**
+ * Fetch code reviews for a repository from CodeRabbit API.
+ * Implements graceful error handling with 30-second timeout.
+ *
+ * @param repoUrl - GitHub repository URL (e.g., "https://github.com/example/repo")
+ * @param apiKey - CodeRabbit API key
+ * @returns Array of ReviewData, empty array on any error
+ */
+export async function fetchReviewsForRepo(
+  repoUrl: string,
+  apiKey: string
+): Promise<ReviewData[]> {
+  // Early return if API key is missing
+  if (!apiKey || apiKey.trim() === '') {
+    console.warn('CodeRabbit API key is empty or null');
+    return [];
+  }
+
+  try {
+    // Create abort controller for timeout enforcement
+    const abortController = new AbortController();
+    const timeoutId = setTimeout(() => {
+      abortController.abort();
+    }, TIMEOUT_MS);
+
+    try {
+      const response = await fetch(API_ENDPOINT, {
+        method: 'POST',
+        headers: {
+          'Content-Type': 'application/json',
+          Authorization: `Bearer ${apiKey}`,
+        },
+        body: JSON.stringify({ repository_url: repoUrl }),
+        signal: abortController.signal,
+      });
+
+      clearTimeout(timeoutId);
+
+      // Check response status
+      if (!response.ok) {
+        console.warn(
+          `CodeRabbit API request failed with status ${response.status}`
+        );
+        return [];
+      }
+
+      // Parse JSON response
+      let data: unknown;
+      try {
+        data = await response.json();
+      } catch {
+        console.warn('Failed to parse CodeRabbit API response as JSON');
+        return [];
+      }
+
+      // Validate response structure
+      if (!isResponseValid(data)) {
+        console.warn('CodeRabbit API response has invalid structure');
+        return [];
+      }
+
+      // Transform and validate reviews
+      const now = new Date();
+      const reviews: ReviewData[] = [];
+
+      for (const review of data.reviews) {
+        // Skip reviews with invalid status
+        if (!isValidReviewStatus(review.review_status)) {
+          continue;
+        }
+
+        // Validate required fields
+        if (
+          typeof review.file_path !== 'string' ||
+          typeof review.comment_count !== 'number' ||
+          review.comment_count < 0
+        ) {
+          continue;
+        }
+
+        // Transform to ReviewData
+        const reviewData: ReviewData = {
+          filePath: normalizeFilePath(review.file_path),
+          reviewStatus: review.review_status,
+          commentCount: review.comment_count,
+          summary: review.summary ?? null,
+          externalReviewUrl: review.external_url ?? null,
+          fetchedAt: now,
+        };
+
+        reviews.push(reviewData);
+      }
+
+      return reviews;
+    } catch (error) {
+      clearTimeout(timeoutId);
+
+      // Check for timeout (AbortError)
+      if (error instanceof Error && error.name === 'AbortError') {
+        console.warn('CodeRabbit API request timed out (30s)');
+        return [];
+      }
+
+      // Any other error
+      console.warn('CodeRabbit API request failed:', error);
+      return [];
+    }
+  } catch (error) {
+    console.warn('Unexpected error in fetchReviewsForRepo:', error);
+    return [];
+  }
+}
+
+/**
+ * Type guard to validate API response structure.
+ */
+function isResponseValid(data: unknown): data is { reviews: CodeRabbitReview[] } {
+  return (
+    typeof data === 'object' &&
+    data !== null &&
+    'reviews' in data &&
+    Array.isArray((data as Record<string, unknown>).reviews)
+  );
+}

+/**
+ * Review status type as a union of string literals.
+ * Represents the state of a code review.
+ */
+export type ReviewStatus =
+  | 'approved'
+  | 'needs-review'
+  | 'issues'
+  | 'not-reviewed';
+
+/**
+ * Normalized review data ready for application use.
+ */
+export interface ReviewData {
+  filePath: string;
+  reviewStatus: ReviewStatus;
+  commentCount: number;
+  summary: string | null;
+  externalReviewUrl: string | null;
+  fetchedAt: Date;
+}
+
+/**
+ * Internal representation of API response from CodeRabbit API.
+ * Uses snake_case as returned by the API.
+ */
+export interface CodeRabbitReview {
+  file_path: string;
+  review_status: ReviewStatus;
+  comment_count: number;
+  summary?: string;
+  external_url?: string;
+}
